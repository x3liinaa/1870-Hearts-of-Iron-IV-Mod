country_event = {
	id = fpw_austria.8
	title = fpw_austria.8.t
	desc = fpw_austria.8.d
	picture = GFX_report_event_gathering_protest
	is_triggered_only = yes
	fire_only_once=yes
	option = {
		name = fpw_austria.8.a
		hidden_effect = {
		}
		if = {
			limit = {
				ENG = {
					has_country_flag  = ENG_agreed_cng_berlin
				}
			}
			183 = { transfer_state_to = ENG}
		} 
		if = {
			limit = {
				SOV = {
					has_country_flag  = SOV_agreed_cng_berlin
				}
			}
			TUR = {
				end_puppet = BUL
				end_puppet = ROM
				end_puppet = SER
			}
			SOV = {
				puppet = {
					target = ROM
					end_wars = no
				}
			}
		} 
		104 = { transfer_state_to = AUS}
		804 = { transfer_state_to = AUS }
		AUS = {
			complete_national_focus = AUS_diplomatic_victory
		}
	}
	option = {
		name = fpw_austria.8.b
		hidden_effect = {
		}
		AUS = {
			create_faction = AUS_congress_of_berlin
			declare_war_on = {
				target = TUR
				type = annex_everything
			}
		}
		if = {
			limit = {
				ENG = {
					has_country_flag  = ENG_agreed_cng_berlin
				}
			}
			ENG  = {
				declare_war_on = {
					target = TUR
					type = annex_everything
				}
			}
			ENG = { add_to_faction = AUS }
		} 
		if = {
			limit = {
				SOV = {
					has_country_flag  = SOV_agreed_cng_berlin
				}
			}
			SOV  = {
				declare_war_on = {
					target = TUR
					type = annex_everything
				}
			}
			SOV = { add_to_faction = AUS }
		} 
		AUS = {
			country_event = {
				id = fpw_austria.9
				days = 30
			}
			complete_national_focus = AUS_war_against_the_ottomans
		}

	}
}
